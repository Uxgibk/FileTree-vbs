```vbs

Option Explicit
' =====================================================
'  目录树工具 Directory Tree Generator
'  - 用法：放到任意文件夹，双击运行
'  - 输出：目录树_文件夹名_YYYY-MM-DD_HH-mm-ss.txt
'  - 不过滤任何文件/目录
'  - 弹窗信息同时写入 txt 开头
'  - 弹窗自动关闭
' =====================================================

' ---------------- 开关 ----------------
Dim AUTO_OPEN, POPUP_TIMEOUT
AUTO_OPEN     = False        ' 生成后是否自动打开 txt：True / False
POPUP_TIMEOUT = 5            ' 弹窗自动关闭秒数
' -------------------------------------

' ---------- 中文字符 ----------
Dim S_MULU, S_SHU, S_WENJIAN, S_WENJIANJIA, S_YI, S_SHENGCHENG
Dim S_LUJING, S_WUFA, S_DUQU, S_HAOSHI, S_MIAO, S_AUTOCLOSE

S_MULU       = ChrW(&H76EE) & ChrW(&H5F55)                 ' 目录
S_SHU        = ChrW(&H6811)                                 ' 树
S_WENJIAN    = ChrW(&H6587) & ChrW(&H4EF6)                 ' 文件
S_WENJIANJIA = ChrW(&H6587) & ChrW(&H4EF6) & ChrW(&H5939)  ' 文件夹
S_YI         = ChrW(&H5DF2)                                 ' 已
S_SHENGCHENG = ChrW(&H751F) & ChrW(&H6210)                 ' 生成
S_LUJING     = ChrW(&H8DEF) & ChrW(&H5F84)                 ' 路径
S_WUFA       = ChrW(&H65E0) & ChrW(&H6CD5)                 ' 无法
S_DUQU       = ChrW(&H8BFB) & ChrW(&H53D6)                 ' 读取
S_HAOSHI     = ChrW(&H8017) & ChrW(&H65F6)                 ' 耗时
S_MIAO       = ChrW(&H79D2)                                 ' 秒
S_AUTOCLOSE  = S_MIAO & ChrW(&H540E) & ChrW(&H81EA) & ChrW(&H52A8) & ChrW(&H5173) & ChrW(&H95ED)  ' 秒后自动关闭

Dim S_TREE, S_CANTREAD
S_TREE     = S_MULU & S_SHU                   ' 目录树
S_CANTREAD = S_WUFA & S_DUQU                  ' 无法读取

' ---------- 运行 ----------
Dim FSO
Set FSO = CreateObject("Scripting.FileSystemObject")

Dim here, folderName, outName, outPath
here       = FSO.GetParentFolderName(WScript.ScriptFullName)
folderName = FSO.GetFileName(here)
outName    = S_TREE & "_" & folderName & "_" & FormatStamp(Now) & ".txt"
outPath    = FSO.BuildPath(here, outName)

Dim gLines, gDirCount, gFileCount
Set gLines = CreateObject("System.Collections.ArrayList")
gDirCount  = 0
gFileCount = 0

Dim tStart, elapsed, elapsedStr
tStart = Timer

Call Walk(here, "")

elapsed = Timer - tStart
If elapsed < 0 Then elapsed = elapsed + 86400
elapsedStr = FmtSec(elapsed)

' ---------- 构造信息头 ----------
Dim infoLine1, infoLine2, infoLine3, infoBlock
infoLine1 = S_WENJIAN & S_YI & S_SHENGCHENG & ": " & outName
infoLine2 = S_LUJING & ": " & outPath
infoLine3 = S_WENJIANJIA & ": " & gDirCount & "   " & S_WENJIAN & ": " & gFileCount & vbCrLf & _
            S_HAOSHI & ": " & elapsedStr & " " & S_MIAO
infoBlock = infoLine1 & vbCrLf & infoLine2 & vbCrLf & infoLine3

' ---------- 写入 txt ----------
Dim treeText, text
If gLines.Count = 0 Then
  treeText = folderName & "/"
Else
  treeText = folderName & "/" & vbCrLf & Join(gLines.ToArray(), vbCrLf)
End If
text = infoBlock & vbCrLf & vbCrLf & treeText & vbCrLf
Call WriteUtf8(outPath, text)

' ---------- 单个弹窗，自动关闭 ----------
Dim popupMsg, shell
popupMsg = infoLine1 & vbCrLf & vbCrLf & infoLine2 & vbCrLf & vbCrLf & infoLine3 & _
           vbCrLf & vbCrLf & POPUP_TIMEOUT & " " & S_AUTOCLOSE

Set shell = CreateObject("WScript.Shell")
shell.Popup popupMsg, POPUP_TIMEOUT, S_TREE, 64

If AUTO_OPEN Then
  shell.Run """" & outPath & """", 1, False
End If

' ==================== 过程 ====================

Function FormatStamp(d)
  FormatStamp = Year(d) & "-" & Pad2(Month(d)) & "-" & Pad2(Day(d)) & "_" & _
                Pad2(Hour(d)) & "-" & Pad2(Minute(d)) & "-" & Pad2(Second(d))
End Function

Function Pad2(n)
  Pad2 = Right("0" & n, 2)
End Function

Function FmtSec(s)
  Dim whole, frac
  whole = Int(s)
  frac  = Int((s - whole) * 100 + 0.5)
  If frac >= 100 Then
    whole = whole + 1
    frac  = 0
  End If
  FmtSec = whole & "." & Right("0" & frac, 2)
End Function

Sub Walk(p, prefix)
  Dim folder, item
  Dim dirList, fileList
  Set dirList  = CreateObject("System.Collections.ArrayList")
  Set fileList = CreateObject("System.Collections.ArrayList")

  On Error Resume Next
  Set folder = FSO.GetFolder(p)
  If Err.Number <> 0 Then
    On Error GoTo 0
    gLines.Add prefix & "\--- [" & S_CANTREAD & "]"
    Exit Sub
  End If
  On Error GoTo 0

  For Each item In folder.SubFolders
    dirList.Add item.Name
  Next
  For Each item In folder.Files
    fileList.Add item.Name
  Next

  dirList.Sort
  fileList.Sort

  Dim total, i, conn, label, isDir, np
  total = dirList.Count + fileList.Count
  If total = 0 Then Exit Sub

  For i = 0 To total - 1
    If i < dirList.Count Then
      isDir = True
      label = dirList(i)
    Else
      isDir = False
      label = fileList(i - dirList.Count)
    End If

    If i = total - 1 Then conn = "\--- " Else conn = "+--- "

    If isDir Then
      gLines.Add prefix & conn & label & "/"
      gDirCount = gDirCount + 1
      If i = total - 1 Then
        np = prefix & "    "
      Else
        np = prefix & "|   "
      End If
      Call Walk(p & "\" & label, np)
    Else
      gLines.Add prefix & conn & label
      gFileCount = gFileCount + 1
    End If
  Next
End Sub

Sub WriteUtf8(path, content)
  Dim stream
  Set stream = CreateObject("ADODB.Stream")
  stream.Type = 2
  stream.Charset = "utf-8"
  stream.Open
  stream.WriteText content
  stream.SaveToFile path, 2
  stream.Close
End Sub
```
